import * as React from "react";
import { loadDietConfig, saveDietConfig } from "../diet/dietAnalyticsData";
import { defaultDietConfig } from "../diet/dietAnalyticsTypes";
import type { DietConfig, ComponentKey } from "../diet/dietAnalyticsTypes";
import { DietConfigurationFields, buttonClass } from "../diet/components/DietAnalyticsComponents";
import { AnalyticsCard, AnalyticsSelect, SegmentedControl } from "../../components/analytics/AnalyticsControls";
import { loadHealthData, saveHealthConfig } from "../health/healthData";
import { emptyConfig, muscleGroups } from "../health/healthTypes";
import type { HealthConfig, HealthData } from "../health/healthTypes";
import { loadSofiaConfig, loadSofiaSnapshot, saveSofiaConfig } from "../health/sofiaData";
import { emptySofiaConfig } from "../health/sofiaTypes";
import type { AlertRule, SofiaConfig, SofiaSnapshot } from "../health/sofiaTypes";
import { SofiaConfigurationFields } from "../health/components/SofiaConfigurationFields";

// ============================================================ //
//                                                              //
//   APPLICATION CONFIGURATION PAGE                                  //
//                                                              //
//   Edits period targets and manual control limits.              //
//   This container owns validation and explicit API saves.      //
//                                                              //
// ============================================================ //
export default function ConfigurationPage() {
  // ====================== //
  //                        //
  //   STATE VARIABLES      //
  //                        //
  // ====================== //
  const [section, setSection] = React.useState("diet");
  const [dietConfig, setDietConfig] = React.useState<DietConfig | null>(null);
  const [dietMessage, setDietMessage] = React.useState("");
  const [dietSaving, setDietSaving] = React.useState(false);
  const user = localStorage.getItem("global/email") || "guest";
  const [data, setData] = React.useState<HealthData | null>(null);
  const [config, setConfig] = React.useState<HealthConfig>(emptyConfig);
  const period = "week" as const;
  const [targetId, setTargetId] = React.useState("steps");
  const [metricId, setMetricId] = React.useState("weight");
  const [target, setTarget] = React.useState("");
  const [lower, setLower] = React.useState("");
  const [upper, setUpper] = React.useState("");
  const [message, setMessage] = React.useState("");
  const [saving, setSaving] = React.useState(false);
  const [sofiaConfig, setSofiaConfig] = React.useState<SofiaConfig | null>(null);
  const [sofiaSnapshot, setSofiaSnapshot] = React.useState<SofiaSnapshot | null>(null);
  const [sofiaSection, setSofiaSection] = React.useState<"alerts" | "report" | "routine">("alerts");
  const [sofiaMessage, setSofiaMessage] = React.useState("");
  const [sofiaSaving, setSofiaSaving] = React.useState(false);
  // ====================== //
  //                        //
  //   OBSERVE STATE        //
  //                        //
  // ====================== //
  console.log("health configuration selection", period, targetId, metricId, saving);
  // ====================== //
  //                        //
  //   SIDE EFFECTS         //
  //                        //
  // ====================== //
  React.useEffect(() => {
    let active = true;
    void loadDietConfig(user).then(config => { if (active) setDietConfig(config); }).catch(error => { const failure = new Error("Unable to load diet_config"); failure.name = error instanceof Error ? error.name : "UnknownError"; console.log("Error loading Diet configuration", failure); if (active) setDietMessage("Unable to load Diet settings."); });
    return () => { active = false; };
  }, [user]);
  React.useEffect(() => {
    let active = true;
    void loadHealthData(user).then(next => { if (active) { setData(next); if (next.config) setConfig(next.config); } });
    return () => { active = false; };
  }, [user]);
  React.useEffect(() => {
    let active = true;
    void Promise.allSettled([loadSofiaConfig(user), loadSofiaSnapshot(user)]).then(results => {
      if (!active) return;
      if (results[0].status === "fulfilled") setSofiaConfig(results[0].value);
      if (results[1].status === "fulfilled") setSofiaSnapshot(results[1].value);
      setSofiaMessage(results.filter(result => result.status === "rejected").map(result => String(result.reason)).join(". "));
    });
    return () => { active = false; };
  }, [user]);
  React.useEffect(() => {
    setTarget(config.targets[targetId]?.[period]?.toString() ?? "");
  }, [config, period, targetId]);
  React.useEffect(() => {
    setLower(config.limits[metricId]?.[period]?.lower.toString() ?? "");
    setUpper(config.limits[metricId]?.[period]?.upper.toString() ?? "");
  }, [config, period, metricId]);
  // ====================== //
  //                        //
  //   UI EVENT HANDLERS    //
  //                        //
  // ====================== //
  const handleEventSection = (value: string) => setSection(value);
  // ------------------------------------------------------ Configuration
  const handleEventTargetId = (value: string) => setTargetId(value);
  const handleEventMetricId = (value: string) => setMetricId(value);
  const handleEventTarget = (value: string) => setTarget(value);
  const handleEventLower = (value: string) => setLower(value);
  const handleEventUpper = (value: string) => setUpper(value);
  const handleEventSave = async (event: React.FormEvent) => {
    event.preventDefault();
    const hasTarget = target.trim() !== "";
    const hasLower = lower.trim() !== "";
    const hasUpper = upper.trim() !== "";
    if ((hasTarget && (!Number.isFinite(Number(target)) || Number(target) < 0)) || (hasLower && !Number.isFinite(Number(lower))) || (hasUpper && !Number.isFinite(Number(upper)))) {
      setMessage("Enter finite values and a non-negative target."); return;
    }
    if (hasLower !== hasUpper || (hasLower && Number(lower) >= Number(upper))) {
      setMessage("Enter both limits with the lower limit less than the upper limit, or leave both empty."); return;
    }
    const next: HealthConfig = { ...config, targets: { ...config.targets, [targetId]: { ...config.targets[targetId] } }, limits: { ...config.limits, [metricId]: { ...config.limits[metricId] } } };
    if (hasTarget) next.targets[targetId][period] = Number(target); else delete next.targets[targetId][period];
    if (hasLower) next.limits[metricId][period] = { lower: Number(lower), upper: Number(upper) }; else delete next.limits[metricId][period];
    setSaving(true);
    try { await saveHealthConfig(user, next); setConfig(next); setData(previous => previous ? { ...previous, config: next } : previous); setMessage("Health configuration saved."); }
    catch (error) { setMessage(error instanceof Error ? error.message : "Unable to save Health configuration."); }
    finally { setSaving(false); }
  };
  // ------------------------------------------------------ Diet configuration
  const handleEventDietRule = (key: ComponentKey, patch: Partial<DietConfig["rules"][ComponentKey]>) => setDietConfig(previous => previous && ({ ...previous, rules: { ...previous.rules, [key]: { ...previous.rules[key], ...patch } } }));
  const handleEventDietCorrection = (correction: number) => setDietConfig(previous => previous && ({ ...previous, correction }));
  const handleEventDietSave = async (event: React.FormEvent) => {
    event.preventDefault();
    if (!dietConfig) return;
    setDietSaving(true);
    try { await saveDietConfig(user, dietConfig); setDietMessage("Diet settings saved."); }
    catch (error) { setDietMessage(error instanceof Error ? error.message : "Unable to save Diet settings."); }
    finally { setDietSaving(false); }
  };
  // ------------------------------------------------------ Sofia configuration
  const handleEventSofiaRuleChange = (id: string, patch: Partial<AlertRule>) => setSofiaConfig(current => current && ({ ...current, rules: current.rules.map(rule => rule.id === id ? { ...rule, ...patch } : rule) }));
  const handleEventSofiaRuleAdd = () => {
    const metric = sofiaSnapshot?.metrics[0];
    if (!metric) return;
    setSofiaConfig(current => current && ({ ...current, rules: [...current.rules, { id: crypto.randomUUID(), source_key: metric.source_key, kind: "pace", enabled: false, direction: "minimum", tolerance: 0, stale_after_days: null, period: "week", include_in_report: true }] }));
  };
  const handleEventSofiaRuleRemove = (id: string) => setSofiaConfig(current => current && ({ ...current, rules: current.rules.filter(rule => rule.id !== id) }));
  const handleEventSofiaWeightChange = (id: string, weight: number) => setSofiaConfig(current => current && ({ ...current, habit_weights: { ...current.habit_weights, [id]: weight } }));
  const handleEventSofiaSection = (section: "alerts" | "report" | "routine") => setSofiaSection(section);
  const handleEventSofiaSave = async (event: React.FormEvent) => {
    event.preventDefault();
    if (!sofiaConfig) return;
    setSofiaSaving(true);
    try { await saveSofiaConfig(user, sofiaConfig); setSofiaMessage("Sofia settings saved"); }
    catch (error) { setSofiaMessage(String(error)); }
    finally { setSofiaSaving(false); }
  };
  // ====================== //
  //                        //
  //   UTILS METHODS        //
  //                        //
  // ====================== //
  const targetOptions = [{ id: "steps", name: "Steps" }, ...muscleGroups.map(id => ({ id: `volume:${id}`, name: `${id[0].toUpperCase() + id.slice(1)} volume` }))];
  const fieldClass = "w-full rounded-lg border border-gray-200 px-3 py-2 text-sm text-gray-800";
  // ====================== //
  //                        //
  //   UI COMPONENTS        //
  //                        //
  // ====================== //
  return <main className="mx-auto max-w-7xl space-y-5 px-4 py-6 text-gray-800">
    <h1 className="text-2xl font-semibold">Configuration</h1>
    <SegmentedControl label="Section" value={section} options={[{ value: "diet", label: "Diet" }, { value: "health", label: "Health" }, { value: "sofia", label: "Sofia" }]} onEventChange={handleEventSection} />
    {section === "diet" && <form onSubmit={handleEventDietSave} className="space-y-5"><DietConfigurationFields config={dietConfig ?? defaultDietConfig()} onEventRule={handleEventDietRule} onEventCorrection={handleEventDietCorrection}/><button type="submit" className={buttonClass} disabled={!dietConfig || dietSaving}>{dietSaving ? "Saving…" : "Save Diet settings"}</button>{dietMessage && <p role="status" className="text-sm text-gray-700">{dietMessage}</p>}</form>}
    {section === "health" && <>
    {data?.errors.map((error, index) => <p role="status" key={index} className="text-sm text-gray-500">{error}</p>)}
    {/* Period configuration */}
    <form onSubmit={handleEventSave} className="space-y-5">
      {/* Burnup target */}
      <AnalyticsCard><h2 className="mb-4 text-sm font-semibold">Burnup target</h2><div className="grid items-end gap-4 sm:grid-cols-2"><AnalyticsSelect label="Activity" value={targetId} options={targetOptions} onEventChange={handleEventTargetId} /><label className="space-y-1 text-xs text-gray-500">Target total<input aria-label="Target total" className={fieldClass} type="number" min={0} step="any" value={target} onChange={event => handleEventTarget(event.target.value)} /></label></div><p className="mt-3 text-xs text-gray-500">Steps use a step count. Muscle volume uses recorded kg × reps. Leave empty to unset.</p></AnalyticsCard>
      {/* Manual control limits */}
      <AnalyticsCard><h2 className="mb-4 text-sm font-semibold">Control limits</h2><div className="grid items-end gap-4 sm:grid-cols-3"><AnalyticsSelect label="Metric" value={metricId} options={data?.analytics?.metrics ?? [{ id: "weight", name: "Weight" }]} onEventChange={handleEventMetricId} /><label className="space-y-1 text-xs text-gray-500">Lower limit<input aria-label="Lower limit" className={fieldClass} type="number" step="any" value={lower} onChange={event => handleEventLower(event.target.value)} /></label><label className="space-y-1 text-xs text-gray-500">Upper limit<input aria-label="Upper limit" className={fieldClass} type="number" step="any" value={upper} onChange={event => handleEventUpper(event.target.value)} /></label></div><p className="mt-3 text-xs text-gray-500">Limits apply to the selected metric and period in its displayed units. Leave both empty to unset.</p></AnalyticsCard>
      {/* Explicit save */}
      <button type="submit" disabled={saving || !data?.config} className="rounded-lg bg-violet-700 px-4 py-2 text-sm font-semibold text-white hover:bg-violet-800 disabled:opacity-50">{saving ? "Saving…" : "Save configuration"}</button>
      {message && <p role="status" className="text-sm text-gray-700">{message}</p>}
    </form>
    </>}
    {section === "sofia" && <AnalyticsCard><h2 className="mb-4 text-sm font-semibold">Sofia settings</h2>
      <SegmentedControl label="Sofia settings section" value={sofiaSection} options={[{ value: "alerts", label: "Alerts" }, { value: "report", label: "Report" }, { value: "routine", label: "Routine" }]} onEventChange={handleEventSofiaSection} />
      <form onSubmit={handleEventSofiaSave} className="mt-5 space-y-5">
        <SofiaConfigurationFields section={sofiaSection} config={sofiaConfig ?? emptySofiaConfig()} snapshot={sofiaSnapshot} onEventRuleChange={handleEventSofiaRuleChange} onEventRuleAdd={handleEventSofiaRuleAdd} onEventRuleRemove={handleEventSofiaRuleRemove} onEventWeightChange={handleEventSofiaWeightChange} />
        <button type="submit" disabled={!sofiaConfig || sofiaSaving} className="rounded-lg bg-violet-700 px-4 py-2 text-sm font-semibold text-white disabled:opacity-50">{sofiaSaving ? "Saving…" : "Save Sofia settings"}</button>
        {sofiaMessage && <p role="status" className="text-sm text-gray-700">{sofiaMessage}</p>}
      </form>
    </AnalyticsCard>}
  </main>;
}
